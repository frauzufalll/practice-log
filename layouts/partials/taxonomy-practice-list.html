{{/* taxonomy-practice-list.html Expects: nodes = [] of { name:string, children: [] } practices = [] of enriched practice dicts */}} {{- $practices := .practices }} {{- $depth := .depth }} {{- range $node := .nodes }}
    {{/* find all practices that include this element */}} {{- $matches := where $practices "elements" "intersect" (slice $node.name) }} {{- if eq $depth 0 }}
  • {{ $node.name }}
    • {{- end }} {{- if gt (len $matches) 0 }} {{- if ne $depth 0 }}
  • {{ $node.name }}
    • {{- end }}
      {{- range $p := $matches }}
    • {{ $p.description }}
      {{ $p.date }}
      • {{- end }}
    {{- end }}
{{/* recurse into any children */}} {{- with $node.children }} {{- partial "taxonomy-practice-list.html" (dict "nodes" . "practices" $practices "depth" (add $depth 1) ) }} {{- end }} {{- end }}
